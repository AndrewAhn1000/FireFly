cmake_minimum_required(VERSION 3.24)
project(FireFlyRuntime LANGUAGES CXX)
find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Lua REQUIRED)
add_library(firefly-observations graph.cpp lua_node.cpp onnx_node.cpp expression.cpp geometry.cpp template_fit.cpp template_match.cpp ocr_node.cpp tracked.cpp)
target_compile_features(firefly-observations PUBLIC cxx_std_20)
target_link_libraries(firefly-observations PUBLIC ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
target_include_directories(firefly-observations PUBLIC ${LUA_INCLUDE_DIR})
target_link_libraries(firefly-observations PUBLIC ${LUA_LIBRARIES})
# Optional: ONNX Runtime pre-built (download with npm run native:onnxruntime)
set(ORT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/libs/onnxruntime")
option(FIREFLY_DIRECTML "Use the GPU-capable DirectML runtime when installed" ON)
if(FIREFLY_DIRECTML AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/libs/onnxruntime-directml/include/onnxruntime_cxx_api.h")
  set(ORT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/libs/onnxruntime-directml")
  target_compile_definitions(firefly-observations PUBLIC FIREFLY_DIRECTML=1)
  target_link_libraries(firefly-observations PUBLIC dxgi d3d12)
endif()
if(EXISTS "${ORT_DIR}/include/onnxruntime_cxx_api.h")
  message(STATUS "ONNX Runtime found at ${ORT_DIR}")
  target_compile_definitions(firefly-observations PUBLIC FIREFLY_ONNX=1)
  target_include_directories(firefly-observations PUBLIC "${ORT_DIR}/include")
  target_link_libraries(firefly-observations PUBLIC "${ORT_DIR}/lib/onnxruntime.lib")
  set(FIREFLY_ORT_DLL "${ORT_DIR}/lib/onnxruntime.dll")
else()
  message(STATUS "ONNX Runtime not found — 'segment' nodes will error at runtime. Run: npm run native:onnxruntime")
  set(FIREFLY_ORT_DLL "")
endif()
target_include_directories(firefly-observations PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
add_executable(firefly-graph graph_cli.cpp)
target_link_libraries(firefly-graph PRIVATE firefly-observations)
if(WIN32)
find_package(unofficial-sqlite3 CONFIG REQUIRED)
add_library(firefly-windows-ocr ocr_windows.cpp)
target_compile_features(firefly-windows-ocr PUBLIC cxx_std_20)
target_compile_definitions(firefly-windows-ocr PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(firefly-windows-ocr PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(firefly-windows-ocr PUBLIC firefly-observations windowsapp)
target_link_libraries(firefly-graph PRIVATE firefly-windows-ocr)
target_compile_definitions(firefly-graph PRIVATE FIREFLY_WINDOWS_OCR)
add_library(firefly-sessions session.cpp recording.cpp input_monitor.cpp)
target_link_libraries(firefly-sessions PUBLIC firefly-observations unofficial::sqlite3::sqlite3 bcrypt)
target_compile_definitions(firefly-sessions PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(firefly-runtime main.cpp capture_windows.cpp detection.cpp tracking.cpp)
target_compile_features(firefly-runtime PRIVATE cxx_std_20)
target_compile_definitions(firefly-runtime PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(firefly-runtime PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(firefly-runtime PRIVATE d3d11 dxgi windowsapp dwmapi firefly-observations firefly-sessions firefly-windows-ocr)
if(FIREFLY_ORT_DLL)
  add_custom_command(TARGET firefly-runtime POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${FIREFLY_ORT_DLL}"
      "$<TARGET_FILE_DIR:firefly-runtime>/onnxruntime.dll"
    COMMENT "Copying onnxruntime.dll")
endif()

add_executable(session-tests session_tests.cpp)
target_link_libraries(session-tests PRIVATE firefly-sessions)
add_executable(firefly-lab-input lab_input.cpp)
target_compile_features(firefly-lab-input PRIVATE cxx_std_20)
add_executable(firefly-input input_guard.cpp)
target_link_libraries(firefly-input PRIVATE firefly-sessions)
endif()
enable_testing()
add_executable(graph-tests graph_tests.cpp)
target_link_libraries(graph-tests PRIVATE firefly-observations)
add_test(NAME observation-graph COMMAND graph-tests)
add_executable(template-tests template_tests.cpp)
target_link_libraries(template-tests PRIVATE firefly-observations)
add_test(NAME template-fit COMMAND template-tests)
if(WIN32)
add_test(NAME recording-session COMMAND session-tests)
endif()

if(WIN32)
add_executable(detection-tests detection_tests.cpp detection.cpp tracking.cpp)
target_link_libraries(detection-tests PRIVATE firefly-observations windowsapp)
add_test(NAME async-detection COMMAND detection-tests)
endif()
# Every executable linking ORT must load the same runtime, including offline tests.
if(FIREFLY_ORT_DLL)
  file(GLOB ORT_RUNTIME_DLLS "${ORT_DIR}/lib/*.dll")
  foreach(t firefly-runtime firefly-graph graph-tests template-tests session-tests firefly-input detection-tests)
    if(TARGET ${t})
      add_custom_command(TARGET ${t} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ORT_RUNTIME_DLLS} "$<TARGET_FILE_DIR:${t}>"
        COMMAND_EXPAND_LISTS)
    endif()
  endforeach()
endif()
